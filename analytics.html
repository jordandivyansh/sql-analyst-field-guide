Funnel conversion

WITH user_steps AS (
  SELECT
    user_id,
    MIN(CASE WHEN event_name = 'visit' THEN event_time END) AS visited_at,
    MIN(CASE WHEN event_name = 'signup' THEN event_time END) AS signed_up_at,
    MIN(CASE WHEN event_name = 'purchase' THEN event_time END) AS purchased_at
  FROM events
  GROUP BY user_id
)
SELECT
  COUNT(*) AS visitors,
  SUM(signed_up_at IS NOT NULL) AS signups,
  SUM(purchased_at IS NOT NULL) AS purchasers,
  100.0 * SUM(purchased_at IS NOT NULL) / NULLIF(COUNT(*), 0) AS visit_to_purchase_pct
FROM user_steps;

Define whether steps must occur in order and within a conversion window. The query above captures first occurrence but does not yet enforce sequence.

Cohort retention

WITH first_month AS (
  SELECT user_id, DATE_FORMAT(MIN(activity_date), '%Y-%m-01') AS cohort_month
  FROM activity
  GROUP BY user_id
),
monthly_activity AS (
  SELECT DISTINCT user_id, DATE_FORMAT(activity_date, '%Y-%m-01') AS active_month
  FROM activity
)
SELECT
  f.cohort_month,
  TIMESTAMPDIFF(MONTH, f.cohort_month, a.active_month) AS month_number,
  COUNT(DISTINCT a.user_id) AS retained_users
FROM first_month f
JOIN monthly_activity a ON a.user_id = f.user_id
GROUP BY f.cohort_month, month_number;

Divide each retained count by month-0 cohort size to calculate retention rate.

Period-over-period growth

WITH monthly AS (
  SELECT DATE_FORMAT(order_date, '%Y-%m-01') AS month,
         SUM(amount) AS revenue
  FROM orders
  GROUP BY month
),
compared AS (
  SELECT *,
         LAG(revenue) OVER (ORDER BY month) AS previous_revenue
  FROM monthly
)
SELECT *,
       100.0 * (revenue - previous_revenue)
       / NULLIF(previous_revenue, 0) AS mom_growth_pct
FROM compared;

Contribution and Pareto analysis

WITH product_revenue AS (
  SELECT product_id, SUM(amount) AS revenue
  FROM order_items
  GROUP BY product_id
)
SELECT
  product_id,
  revenue,
  100.0 * revenue / SUM(revenue) OVER () AS share_pct,
  100.0 * SUM(revenue) OVER (
    ORDER BY revenue DESC
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  ) / SUM(revenue) OVER () AS cumulative_share_pct
FROM product_revenue;

RFM customer segmentation

DimensionMeaningTypical measure
RecencyHow recently the customer purchasedDays since last order
FrequencyHow often they purchaseDistinct order count
MonetaryHow much value they generatedPaid revenue
SELECT
  customer_id,
  DATEDIFF(CURRENT_DATE, MAX(order_date)) AS recency_days,
  COUNT(DISTINCT order_id) AS frequency,
  SUM(amount) AS monetary_value
FROM orders
WHERE status = 'paid'
GROUP BY customer_id;
